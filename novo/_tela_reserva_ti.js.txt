// Gestor · Cliente (novo) — a TELA (TI, 27/10/2026; casca de reserva com Tabler até a casca do PICASSO chegar — o contrato é o
// mesmo: desenhar(DADOS), onFiltro(filtro), onTratar(ref, cor), onBro()). Blocos do escopo mínimo de hoje: topo com agenda · o que
// decidir · carteira e UMA cobertura com a base · sell-in × sell-out × meta × estoque por ano (ECharts) · Bro (botão → painel IA).
(() => {
  const $ = id => document.getElementById(id);
  const esc = s => String(s == null ? "" : s).replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
  const n0 = v => v == null || isNaN(v) ? "—" : Math.round(v).toLocaleString("pt-BR");
  const n1 = v => v == null || isNaN(v) ? "—" : (+v).toLocaleString("pt-BR", { minimumFractionDigits: 1, maximumFractionDigits: 1 });
  const dmy = s => s ? String(s).slice(0, 10).split("-").reverse().join("/") : "—";
  const MESES = ["jan", "fev", "mar", "abr", "mai", "jun", "jul", "ago", "set", "out", "nov", "dez"];
  const AZUL = "#2E3EFC", AMARELO = "#FFFF1A", PRETO = "#151515", CINZA = "#787B86", CINZA2 = "#ADAEB3";
  const CORES_ANO = { "2024": CINZA2, "2025": CINZA, "2026": AZUL, "2027": PRETO };
  let CLIENTE = null, DADOS = null, grafico = null, camadas = { sellin: true, sellout: true, meta: true, estoque: true };

  // ---------- arranque: login → clientes → cliente da URL ou o primeiro
  async function arranque() {
    const s = await Dados.sessao();
    if (!s) { $("login").classList.remove("d-none"); $("form-login").onsubmit = async ev => { ev.preventDefault(); $("login-msg").textContent = ""; const e = await Dados.entrar($("email").value.trim(), $("senha").value); if (e) $("login-msg").textContent = "não entrou: " + e; else location.reload(); }; return; }
    $("app").classList.remove("d-none"); $("bt-sair").onclick = ev => { ev.preventDefault(); Dados.sair(); };
    if (!Dados.MODO_CLIENTE) { $("bt-bro").classList.remove("d-none"); $("bt-bro").onclick = onBro; }
    const lista = await Dados.clientes(), h = new URLSearchParams(location.hash.slice(1));
    const sel = $("sel-cliente"); sel.innerHTML = lista.map(c => `<option value="${esc(c.cliente)}">${esc(c.nome || c.cliente)}</option>`).join("");
    CLIENTE = lista.some(c => c.cliente === h.get("cliente")) ? h.get("cliente") : (lista.some(c => c.cliente === "BANBAN CALCADOS") ? "BANBAN CALCADOS" : (lista[0] || {}).cliente);
    if (Dados.MODO_CLIENTE) { sel.disabled = true; $("bt-gestor").classList.add("d-none"); }        // a página do cliente não lista outros clientes
    sel.value = CLIENTE; sel.onchange = () => { CLIENTE = sel.value; location.hash = "cliente=" + encodeURIComponent(CLIENTE); abrir(); };
    abrir();
  }
  async function abrir() {
    $("tela").innerHTML = `<div class="text-secondary py-5 text-center">carregando ${esc(CLIENTE)}…</div>`;
    try { DADOS = await Dados.carregar(CLIENTE); desenhar(DADOS); }
    catch (e) { $("tela").innerHTML = `<div class="alert alert-danger">não consegui ler o banco: ${esc(e.message || e)}</div>`; }
  }
  function onFiltro(filtro) { DADOS = Dados.refiltrar(filtro); desenharGrafico(DADOS); desenharFiltros(DADOS); }
  function onTratar(ref, cor) { window.open(`../#aba=cg&cliente=${encodeURIComponent(CLIENTE)}&ref=${encodeURIComponent(ref)}${cor ? "&cor=" + encodeURIComponent(cor) : ""}`, "_blank"); }
  function onBro() { window.open(`../#aba=cliente&cliente=${encodeURIComponent(CLIENTE)}&ia=1`, "_blank"); }   // o painel IA (Bro) vive no Gestor; a gaveta própria vem no porte

  // ---------- a tela
  function desenhar(d) {
    const c = d.cliente, cob = d.giro.cobertura;
    const agenda = d.agenda.length ? d.agenda.map(a => `<div><span class="text-secondary">${dmy(a.data)}</span> · ${esc(a.titulo)}</div>`).join("") : `<span class="text-secondary">sem reunião registrada</span>`;
    const decidir = d.decidir.length ? d.decidir.map(i => `<div class="item-decidir${i.urgente ? " urgente" : ""}"><div>${esc(i.texto)}</div><div class="quando">${i.sub ? esc(i.sub) + " · " : ""}${i.prazo ? "prazo " + dmy(i.prazo) : (i.dono ? "com " + esc(i.dono) : "")}</div></div>`).join("") : `<div class="text-secondary">nada pendente</div>`;
    const cartMeses = d.carteira.meses.slice(-6);
    $("tela").innerHTML = `
      <div class="row row-cards">
        <div class="col-12"><div class="card"><div class="card-body d-flex flex-wrap align-items-center gap-3">
          <div><div class="h1 mb-0" style="font-family:Barlow;font-weight:800">${esc(c.nome)}</div><div class="text-secondary small">${c.lojas_ativas ? c.lojas_ativas + " lojas ativas · " : ""}${esc(c.recorte || "")}</div></div>
          <div class="ms-auto small"><div class="text-uppercase text-secondary" style="letter-spacing:.08em;font-size:.7rem">Agenda · últimas reuniões</div>${agenda}</div>
        </div></div></div>
        <div class="col-md-5"><div class="card h-100"><div class="card-header"><h2 class="card-title">O que decidir hoje</h2><span class="ms-auto badge badge-oly">${d.decidir.length} pendente${d.decidir.length === 1 ? "" : "s"}</span></div><div class="card-body">${decidir}</div></div></div>
        <div class="col-md-7"><div class="card h-100"><div class="card-header"><h2 class="card-title">Carteira e cobertura</h2><span class="ms-auto etiqueta">${esc(d.giro.recorte)}</span></div><div class="card-body">
          <div class="kpi mb-3">
            <div><b>${n0(d.carteira.a_faturar)}</b><small>carteira a faturar · base ${dmy(d.carteira.base_em)}</small></div>
            <div><b>${d.giro.estoque != null ? n0(d.giro.estoque) : "—"}</b><small>estoque · ${d.giro.estoque_mes ? Dados.mesBR(d.giro.estoque_mes) : "sem estoque no banco"}</small></div>
            <div class="${cob != null && cob < 4 ? "alerta" : ""}"><b>${cob != null ? n1(cob) : "—"}</b><small>cobertura · meses</small></div>
          </div>
          <div class="etiqueta mb-3">${esc(d.giro.cobertura_base)} · alerta abaixo de 4 meses</div>
          <div class="barras">${d.giro.serie.map(x => `<div class="${x.parcial ? "parcial" : ""}" style="height:${Math.round(100 * x.v / Math.max(1, ...d.giro.serie.map(y => y.v)))}%" title="${Dados.mesBR(x.mes)}: ${n0(x.v)}"><span>${n0(x.v)}</span></div>`).join("")}</div>
          <div class="eixo">${d.giro.serie.map(x => `<span>${Dados.mesBR(x.mes)}${x.parcial ? "*" : ""}</span>`).join("")}</div>
          ${cartMeses.length ? `<table class="table table-sm table-borderless mt-3 mb-0 small"><thead><tr><th>carteira</th>${cartMeses.map(x => `<th class="text-end">${Dados.mesBR(x.mes)}</th>`).join("")}</tr></thead>
            <tbody><tr><td>faturado</td>${cartMeses.map(x => `<td class="text-end">${x.fat ? n0(x.fat) : "—"}</td>`).join("")}</tr><tr><td>em carteira</td>${cartMeses.map(x => `<td class="text-end">${x.cart ? n0(x.cart) : "—"}</td>`).join("")}</tr></tbody></table>` : ""}
        </div></div></div>
        <div class="col-12"><div class="card"><div class="card-header"><h2 class="card-title">Sell-in × sell-out × meta · por ano</h2><span class="ms-auto etiqueta" id="unidade"></span></div>
          <div class="card-body"><div class="filtros" id="filtros"></div><div id="grafico"></div>
          <div class="etiqueta mt-2">${esc(d.etiquetas.sellin)} · ${esc(d.etiquetas.sellout)} · ${esc(d.etiquetas.meta)} · ${esc(d.etiquetas.estoque)} · clique na legenda para ligar e desligar</div></div></div></div>
      </div>`;
    $("rodape").textContent = `carteira: base ${dmy(d.atualizado.carteira)} · sell-in: puxado em ${dmy(d.atualizado.sellin)} · só dado do banco, depois do login`;
    desenharFiltros(d); desenharGrafico(d);
  }

  // ---------- filtros em blocos lado a lado (palavra dele, 27/10): marca › tipo › gênero e categorias do tênis; múltipla escolha
  function desenharFiltros(d) {
    const f = d.filtro, el = $("filtros"); if (!el) return;
    const bt = (grupo, v, rot, ativo) => `<button type="button" class="btn btn-sm${ativo ? " ativo" : ""}" data-g="${grupo}" data-v="${esc(v)}">${esc(rot)}</button>`;
    const tipos = [["tenis", "tênis"], ["meias", "meias"], ["chinelo", "chinelo"], ["acessorio", "acessório"], ["confeccao", "confecção"]];
    const cats = [["CONF_MASC", "conforto masc."], ["CONF_FEM", "conforto fem."], ["CORRIDA/TREINO", "corrida/treino"], ["CORRE", "corre"], ["CLASSICOS", "clássicos"], ["INFANTIL", "infantil"]];
    el.innerHTML = `<span class="bloco"><span class="rot">marca</span>${bt("marca", "OLY", "Olympikus", f.marca.includes("OLY"))}${bt("marca", "UA", "Under Armour", f.marca.includes("UA"))}</span>`
      + `<span class="bloco"><span class="rot">tipo</span>${tipos.map(([v, r]) => bt("tipo", v, r, f.tipo === v)).join("")}</span>`
      + (f.tipo === "tenis" ? `<span class="bloco"><span class="rot">gênero</span>${["MASCULINO", "FEMININO", "INFANTIL"].map(g => bt("genero", g, g.toLowerCase(), f.genero.includes(g))).join("")}</span>`
      + `<span class="bloco"><span class="rot">categoria</span>${cats.map(([v, r]) => bt("categoria", v, r, f.categoria.includes(v))).join("")}</span>` : "");
    for (const b of el.querySelectorAll("button")) b.onclick = () => {
      const g = b.dataset.g, v = b.dataset.v, nf = JSON.parse(JSON.stringify(f));
      if (g === "tipo") { nf.tipo = v; nf.genero = []; nf.categoria = []; }
      else if (g === "marca") { nf.marca = nf.marca.includes(v) ? nf.marca.filter(x => x !== v) : [...nf.marca, v]; if (!nf.marca.length) nf.marca = [v]; }
      else if (g === "genero") { nf.genero = nf.genero.includes(v) ? nf.genero.filter(x => x !== v) : [...nf.genero, v]; nf.categoria = []; }   // gênero limpa categoria e vice-versa
      else { nf.categoria = nf.categoria.includes(v) ? nf.categoria.filter(x => x !== v) : [...nf.categoria, v]; nf.genero = []; }
      onFiltro(nf);
    };
  }

  // ---------- o gráfico: uma linha por ano e camada (sell-in cheia, sell-out tracejada, meta pontilhada); estoque em área no eixo da direita
  function desenharGrafico(d) {
    const el = $("grafico"); if (!el) return;
    if (!grafico) { grafico = echarts.init(el, null, { renderer: "svg" }); window.addEventListener("resize", () => grafico && grafico.resize()); }
    const A = d.anos, series = [], legend = [];
    const cor = a => CORES_ANO[a] || PRETO;
    for (const a of A.lista) {
      const add = (k, nome, tipo, opts) => { if (!(A[k] && A[k][a]) || A[k][a].every(v => v == null)) return; legend.push(nome); series.push(Object.assign({ name: nome, type: tipo, data: A[k][a], smooth: false, symbol: "circle", symbolSize: 5, color: cor(a), connectNulls: false }, opts)); };
      add("sellin", `sell-in ${a}`, "line", { lineStyle: { width: 3 } });
      add("sellout", `sell-out ${a}`, "line", { lineStyle: { width: 2, type: "dashed" } });
      add("meta", `meta ${a}`, "line", { lineStyle: { width: 2, type: "dotted" }, symbol: "diamond" });
      add("estoque", `estoque ${a}`, "line", { yAxisIndex: 1, areaStyle: { opacity: .08 }, lineStyle: { width: 1 }, symbol: "none", z: 1, silent: true });
    }
    $("unidade").textContent = A.unidade;
    grafico.setOption({
      color: [AZUL], animationDuration: 500,
      tooltip: { trigger: "axis", valueFormatter: v => v == null ? "—" : Math.round(v).toLocaleString("pt-BR") },
      legend: { data: legend, type: "scroll", bottom: 0, textStyle: { color: PRETO } },
      grid: { left: 56, right: 56, top: 24, bottom: 56 },
      xAxis: { type: "category", data: MESES, axisLine: { lineStyle: { color: CINZA2 } }, axisLabel: { color: CINZA } },
      yAxis: [{ type: "value", name: A.unidade, axisLabel: { color: CINZA, formatter: v => v.toLocaleString("pt-BR") }, splitLine: { lineStyle: { color: "#eee" } } },
              { type: "value", name: "estoque", alignTicks: true, splitLine: { show: false }, axisLabel: { color: CINZA2, formatter: v => v.toLocaleString("pt-BR") } }],
      series
    }, { replaceMerge: ["series", "legend"] });
  }

  window.Tela = { desenhar, onFiltro, onTratar, onBro };
  arranque();
})();
